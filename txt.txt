3. ASYNCHRONOUS QUEUE PRODUCER & WORKER SUBSYSTEM

A. Queue Producer Engine (modules/AsyncQueue.php)

<?php
declare(strict_types=1);

class AsyncQueue {

    /**
     * Enqueue a job for non-blocking asynchronous background execution
     */
    public static function push(int $userId, string $jobType, array$payload, ?string $scheduledAt = null): bool {$db = Database::getInstance();
        $stmt =$db->prepare("INSERT INTO async_job_queue (user_id, job_type, payload, status, scheduled_at) VALUES (?, ?, ?, 'pending', ?)");
        return $stmt->execute([
            $userId,$jobType,
            json_encode($payload, JSON_UNESCAPED_SLASHES \vert{} JSON_UNESCAPED_UNICODE),$scheduledAt ?? date('Y-m-d H:i:s')
        ]);
    }
}


B. Unified Third-Party Gateway (modules/ThirdPartyGateway.php)

<?php
declare(strict_types=1);

require_once __DIR__ . '/../config/site_setting_and_config.php';

class ThirdPartyGateway {

    private static function getGatewayConfig(string $provider): array {$db = Database::getInstance();
        $stmt =$db->prepare("SELECT api_key, base_url FROM third_party_gateways WHERE provider_key = ? AND is_active = 1");
        $stmt->execute([$provider]);
        $config =$stmt->fetch(PDO::FETCH_ASSOC);

        if (!$config) {
            throw new Exception("Gateway provider '{$provider}' is disabled or unconfigured.");
        }
        return $config;
    }

    public static function executeCurl(string $provider, string $endpoint, string$method = 'POST', array $body = []): array {$config = self::getGatewayConfig($provider);$url = rtrim($config['base_url'], '/') . '/' . ltrim($endpoint, '/');

        $ch = curl_init($url);
        curl_setopt_array($ch, [
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_CUSTOMREQUEST => strtoupper($method),
            CURLOPT_TIMEOUT => 30,
            CURLOPT_HTTPHEADER => [
                'Authorization: Bearer ' . $config['api_key'],
                'Content-Type: application/json',
                'Accept: application/json'
            ]
        ]);

        if (!empty($body) && in_array(strtoupper($method), ['POST', 'PUT', 'PATCH'], true)) {
            curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($body));
        }

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $error = curl_error($ch);
        curl_close($ch);

        if ($error) {
            throw new Exception("cURL Error [{$provider}]: " . $error);
        }

        $decoded = json_decode($response, true);
        return [
            'success' => ($httpCode >= 200 && $httpCode < 300), 'status_code' =>$httpCode,
            'data' => $decoded ?? $response
        ];
    }

    // --- Connection Handshakes (Replacing Native OAuth) ---
    public static function getBundleConnectUrl(int $userId, string$platform): string {
        $config = self::getGatewayConfig('bundle_social');$redirectUrl = BASE_URL . "/dashboard?action=bundle_connected";
        return $config['base_url'] . "/connect/{$platform}?redirect_uri=" . urlencode($redirectUrl) . "&state={$userId}";
    }

    public static function getZernioAdConnectUrl(int $userId, string$network): string {
        $config = self::getGatewayConfig('zernio');$redirectUrl = BASE_URL . "/dashboard?action=zernio_connected";
        return $config['base_url'] . "/connect/{$network}?redirect_uri=" . urlencode($redirectUrl) . "&state={$userId}";
    }
}


C. CLI Asynchronous Queue Worker Daemon (cron/queue_worker.php)

<?php
declare(strict_types=1);

if (php_sapi_name() !== 'cli') {
    http_response_code(403);
    die("CLI Execution Required");
}

require_once __DIR__ . '/../config/site_setting_and_config.php';
require_once __DIR__ . '/../modules/ThirdPartyGateway.php';

$db = Database::getInstance();
echo "[Worker] Starting Async Queue Daemon...\n";

while (true) {
    // Fetch pending jobs using SELECT FOR UPDATE lock
    $db->beginTransaction();
    $stmt =$db->prepare("SELECT * FROM async_job_queue WHERE status = 'pending' AND scheduled_at <= NOW() ORDER BY id ASC LIMIT 10 FOR UPDATE");
    $stmt->execute();
    $jobs =$stmt->fetchAll(PDO::FETCH_ASSOC);

    if (empty($jobs)) {$db->commit();
        sleep(2); // Rest worker cycle
        continue;
    }

    // Mark as processing
    $jobIds = array_column($jobs, 'id');$inClause = implode(',', array_map('intval', $jobIds));$db->exec("UPDATE async_job_queue SET status = 'processing' WHERE id IN ({$inClause})");
    $db->commit();

    foreach ($jobs as$job) {
        $jobId = (int)$job['id'];
        $payload = json_decode($job['payload'], true);
        echo "[Worker] Processing Job #{$jobId} - Type: {$job['job_type']}\n";

        try {
            switch ($job['job_type']) {

                // --- BUNDLE.SOCIAL ORGANIC POST DISPATCH ---
                case 'bundle_publish_post':
                    $res = ThirdPartyGateway::executeCurl('bundle_social', '/posts', 'POST', [
                        'account_ids' => $payload['account_ids'],
                        'text' => $payload['caption'],
                        'media' => $payload['media_urls'] ?? [],
                        'scheduled_at' => $payload['scheduled_at'] ?? null
                    ]);

                    if (!$res['success']) {
                        throw new Exception("Bundle.social Publish Error: " . json_encode($res['data']));
                    }

                    $bundlePostId =$res['data']['id'] ?? null;
                    $db->prepare("UPDATE scheduled_posts SET status = 'published', bundle_post_id = ? WHERE id = ?")
                       ->execute([$bundlePostId,$payload['internal_post_id']]);
                    break;

                // --- ZERNIO PAID AD CAMPAIGN LAUNCH ---
                case 'zernio_launch_ad':
                    $res = ThirdPartyGateway::executeCurl('zernio', '/campaigns', 'POST', [
                        'ad_account_id' => $payload['ad_account_id'],
                        'platforms' => $payload['platforms'],
                        'campaign' => $payload['campaign_spec']
                    ]);

                    if (!$res['success']) {
                        throw new Exception("Zernio Campaign Launch Error: " . json_encode($res['data']));
                    }

                    $zernioCampaignId =$res['data']['campaign_id'] ?? null;
                    $db->prepare("UPDATE ad_campaigns SET status = 'active', zernio_campaign_id = ? WHERE id = ?")
                       ->execute([$zernioCampaignId,$payload['internal_campaign_id']]);
                    break;

                // --- ZERNIO AD STATUS TOGGLE (PAUSE/RESUME) ---
                case 'zernio_toggle_status':
                    $res = ThirdPartyGateway::executeCurl('zernio', "/campaigns/{$payload['zernio_campaign_id']}/status", 'PATCH', [
                        'status' => $payload['target_status']
                    ]);

                    if (!$res['success']) {
                        throw new Exception("Zernio Status Toggle Error: " . json_encode($res['data']));
                    }

                    $db->prepare("UPDATE ad_campaigns SET status = ? WHERE id = ?")
                       ->execute([$payload['target_status'],$payload['internal_campaign_id']]);
                    break;

                default:
                    throw new Exception("Unknown job type: " . $job['job_type']);
            }

            // Mark Job Completed
            $db->prepare("UPDATE async_job_queue SET status = 'completed', processed_at = NOW() WHERE id = ?")->execute([$jobId]);
            echo "[Worker] Job #{$jobId} COMPLETED.\n";

        } catch (Exception $e) {
            $attempts = (int)$job['attempts'] + 1;
            $maxAttempts = (int)$job['max_attempts'];$newStatus = ($attempts >=$maxAttempts) ? 'failed' : 'pending';

            $db->prepare("UPDATE async_job_queue SET status = ?, attempts = ?, error_log = ? WHERE id = ?")
               ->execute([$newStatus,$attempts, $e->getMessage(),$jobId]);

            echo "[Worker] Job #{$jobId} FAILED (Attempt {$attempts}/{$maxAttempts}): {$e->getMessage()}\n";
        }
    }
}


4. APPLICATION ENDPOINT REFACTORING

A. Non-Blocking Campaign Launch Endpoint (create-ad.php)

<?php
declare(strict_types=1);
require_once __DIR__ . '/config/site_setting_and_config.php';
require_once __DIR__ . '/includes/auth_guard.php';
require_once __DIR__ . '/modules/AsyncQueue.php';

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action_launch_ad'])) {
    csrf_verify();
    
    $userId = (int)$_SESSION['user_id'];
    $title = trim($_POST['title']);
    $type =$_POST['campaign_type'];
    $budget = (float)$_POST['budget_amount'];
    $headline = trim($_POST['headline']);
    $text = trim($_POST['primary_text']);
    $destination = trim($_POST['destination_url']);
    $mediaUrl = trim($_POST['creative_media_url']);
    $adAccountId = trim($_POST['zernio_ad_account_id']);

    $db = Database::getInstance();
    
    // 1. Insert Campaign into DB with 'pending' state
    $stmt =$db->prepare("INSERT INTO ad_campaigns (user_id, title, campaign_type, objective, budget_type, budget_amount, targeting_spec, headline, primary_text, destination_url, creative_media_url, zernio_ad_account_id, status) VALUES (?, ?, ?, 'CONVERSIONS', 'daily', ?, '{}', ?, ?, ?, ?, ?, 'pending')");
    $stmt->execute([$userId, $title,$type, $budget,$headline, $text,$destination, $mediaUrl,$adAccountId]);
    $campaignId = (int)$db->lastInsertId();

    // 2. Enqueue Job for Async Execution via Zernio Gateway
    $platforms = ($type === 'combo_pack') ? ['meta', 'tiktok', 'google', 'snapchat'] : [$type];
    
    AsyncQueue::push($userId, 'zernio_launch_ad', [
        'internal_campaign_id' => $campaignId,
        'ad_account_id' => $adAccountId,
        'platforms' => $platforms,
        'campaign_spec' => [
            'name' => $title,
            'budget' => $budget,
            'creative' => [
                'headline' => $headline,
                'body' => $text,
                'destination_url' => $destination,
                'media_url' => str_starts_with($mediaUrl, 'http') ? $mediaUrl : BASE_URL . '/' . ltrim($mediaUrl, '/')
            ]
        ]
    ]);

    // 3. Return Instant Response to UI
    $_SESSION['flash_msg'] = "Campaign queued successfully! Background workers are provisioning the campaign on Zernio.";
    header('Location: ' . BASE_URL . '/dashboard');
    exit;
}

<?php
declare(strict_types=1);
require_once __DIR__ . '/config/site_setting_and_config.php';
require_once __DIR__ . '/includes/auth_guard.php';
require_once __DIR__ . '/modules/AsyncQueue.php';

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action_schedule_post'])) {

    $platform =$_POST['platform'];
    $caption = trim($_POST['caption']);
    $scheduledTime =$_POST['scheduled_time'];
    $mediaUrl =$_POST['media_url'] ?? '';
    $accountIds =$_POST['bundle_account_ids'] ?? []; // Array of Bundle.social connected profile IDs

    $db = Database::getInstance();

    // 1. Store scheduled post locally
    $stmt =$db->prepare("INSERT INTO scheduled_posts (user_id, platform, post_type, caption, media_url, media_type, scheduled_time, status) VALUES (?, ?, 'manual', ?, ?, 'image', ?, 'scheduled')");
    $stmt->execute([$userId, $platform,$caption, $mediaUrl,$scheduledTime]);
    $postId = (int)$db->lastInsertId();

    // 2. Enqueue Async Job for Bundle.social Dispatch
    AsyncQueue::push($userId, 'bundle_publish_post', [
        'internal_post_id' => $postId,
        'account_ids' => $accountIds,
        'caption' => $caption,
        'media_urls' => !empty($mediaUrl) ? [BASE_URL . '/' . ltrim($mediaUrl, '/')] : [],

    ], $scheduledTime);

    $_SESSION['flash_msg'] = "Post queued! Bundle.social will broadcast it at the scheduled time.";

}
